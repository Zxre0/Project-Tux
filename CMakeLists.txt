cmake_minimum_required(VERSION 3.22)
project(FortniteOnLinux VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network)
find_package(tomlplusplus 3.3 REQUIRED)
find_package(Vulkan REQUIRED)
find_package(ZLIB REQUIRED)
add_library(fol_core
    src/core.cpp src/fortnite.cpp src/runtime.cpp src/prefix_backup.cpp src/platform_linux.cpp
    src/diagnostics.cpp src/launcher.cpp src/reboot.cpp src/proton.cpp src/supervisor.cpp src/manifest.cpp profiles/profiles.qrc)
target_include_directories(fol_core PUBLIC include PRIVATE ${Vulkan_INCLUDE_DIRS})
target_link_libraries(fol_core PUBLIC Qt6::Core Qt6::Network tomlplusplus::tomlplusplus)
target_link_libraries(fol_core PRIVATE ZLIB::ZLIB)
target_compile_options(fol_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(fortnite-linux cli/main.cpp)
target_link_libraries(fortnite-linux PRIVATE fol_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(runtime-fixture tests/runtime_fixture.cpp)
    target_link_libraries(runtime-fixture PRIVATE Qt6::Core)
    add_executable(core-tests tests/core_tests.cpp)
    target_link_libraries(core-tests PRIVATE fol_core)
    add_dependencies(core-tests fortnite-linux runtime-fixture)
    add_test(NAME core COMMAND core-tests)
    add_executable(network-tests tests/network_tests.cpp)
    target_link_libraries(network-tests PRIVATE fol_core)
    add_test(NAME network COMMAND network-tests)
    set_tests_properties(network PROPERTIES LABELS "requires-network-sockets")
    add_executable(ipc-tests tests/ipc_tests.cpp)
    target_link_libraries(ipc-tests PRIVATE fol_core)
    add_test(NAME ipc COMMAND ipc-tests)
    set_tests_properties(ipc PROPERTIES LABELS "requires-local-ipc")
    add_executable(manifest-tests tests/manifest_tests.cpp)
    target_link_libraries(manifest-tests PRIVATE fol_core)
    add_test(NAME manifest COMMAND manifest-tests)
    add_test(NAME cli-help COMMAND fortnite-linux --help)
    add_test(NAME cli-version COMMAND fortnite-linux version)
endif()
install(TARGETS fortnite-linux RUNTIME DESTINATION bin)
